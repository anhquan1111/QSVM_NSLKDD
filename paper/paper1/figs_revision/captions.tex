\begin{figure*}[t]
  \centering
  \includegraphics[width=\textwidth]{figs_revision/fig5_c1_dimension_selection.pdf}
  \caption{%
    \textbf{The three-stage dimension-selection rule, applied unchanged to two
    datasets.} Rows: NSL-KDD (top), UNSW-NB15 (bottom). Columns: (a) cumulative
    explained variance $V(n)$, (b) kernel--target alignment, (c) CNOT-weighted
    cost $Q(n)$. Stage~1 discards $V(n)<0.85$ (shaded), stage~2 keeps
    $\mathrm{KTA}\ge0.95\,\mathrm{KTA}_{\max}$ (open circles), stage~3 takes the
    cheapest survivor (filled). The same rule returns $n^{\ast}=4$ and
    $n^{\ast}=6$.}
  \label{fig:c1-selection}
\end{figure*}

\begin{figure*}[t]
  \centering
  \includegraphics[width=\textwidth]{figs_revision/fig9_learning_curve_nslkdd.pdf}
  \caption{%
    \textbf{Rare-attack enrichment, not sample size alone, decides whether the
    quantum kernel leads on NSL-KDD.} (a,~b) Mean macro-$F_1$ over ten runs on
    the full KDDTest+ set with $95\%$ bands; (c,~d) paired differences against
    each baseline, circled where the Holm-adjusted $p<0.05$. Under the natural
    prior the ordering reverses between $N=2000$ and $N=5000$ (band in c).
    Twelvefold rare enrichment removes the reversal (d).}
  \label{fig:learning-curve}
\end{figure*}

\begin{figure*}[t]
  \centering
  \includegraphics[width=\textwidth]{figs_revision/fig10_regime_map.pdf}
  \caption{%
    \textbf{Regime map: 110 controlled comparisons of \QSVM{} against six
    baselines.} Each cell is one paired comparison over ten runs, Holm-corrected
    within its baseline family. Verdicts are encoded by both colour and glyph so
    the figure survives greyscale printing. The distribution is the result:
    $21$ favourable, $21$ classical-favourable, $68$ inconclusive. A single
    aggregate number would hide exactly the structure this paper is about.}
  \label{fig:regime-map}
\end{figure*}

\begin{figure*}[t]
  \centering
  \includegraphics[width=\textwidth]{figs_revision/fig11_unsw_transfer.pdf}
  \caption{%
    \textbf{What transfers to UNSW-NB15, and what does not.} (a) Mean
    macro-$F_1$ over ten runs on the official test set; (b) paired differences,
    circled where the Holm-adjusted $p<0.05$. The selection rule and the
    entanglement ablation transfer; the sample-complexity crossover does not.
    Against both tree ensembles the difference keeps its sign at every $N$,
    the opposite of NSL-KDD.}
  \label{fig:unsw-transfer}
\end{figure*}

    score.} (a) Kernel--target alignment for the two feature maps, paired within

  \caption{%
    \textbf{Widening the circuit destroys the quantum kernel, and the kernel
    geometry says why.} (a) The sample-complexity sweep re-run at $K=80$, where
    the rule returns $n^{\ast}=8$: all $48$ paired comparisons are
    classical-favourable. (b) Off-diagonal Gram dispersion against width, fitted
    as $n^{-\alpha}$; the \ZZ{} map concentrates about twice as fast as the
    \Zmap{} map. (c) That dispersion predicts macro-$F_1$ for \ZZ{}
